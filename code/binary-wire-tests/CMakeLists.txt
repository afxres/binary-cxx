cmake_minimum_required(VERSION 3.28)
project("binary-wire-tests")
set(CMAKE_CXX_STANDARD 20)
if(POLICY CMP0167)
    cmake_policy(SET CMP0167 NEW)
endif()
find_package(Boost REQUIRED)
include_directories(${Boost_INCLUDE_DIRS})
file(GLOB_RECURSE SOURCE_FILES "src/*.cpp")
add_executable(${PROJECT_NAME} ${SOURCE_FILES})
target_link_libraries(${PROJECT_NAME} "binary" "binary-experimental")
target_compile_definitions(${PROJECT_NAME} PRIVATE "BINARY_WIRE_TESTS_FIXTURE=\"${CMAKE_CURRENT_SOURCE_DIR}/fixtures/wire.tsv\"")
if(BINARY_HAS_INT128)
    target_compile_definitions(${PROJECT_NAME} PRIVATE "BINARY_HAS_INT128")
endif()
